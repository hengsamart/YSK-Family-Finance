ការវិភាគសុខភាពហិរញ្ញវត្ថុគ្រួសារ

ប្រព័ន្ធឆ្លាតវៃធ្វើការវិភាគទិន្នន័យចំណូល ចំណាយ បំណុល និងគោលដៅសន្សំ ដើម្បីផ្ដល់នូវអនុសាសន៍ល្អបំផុតសម្រាប់អនាគតគ្រួសាររបស់អ្នក។

គោលការណ៍ណែនាំ 50/30/20

៥០% លើតម្រូវការចាំបាច់ (Needs): ថ្លៃផ្ទះ ម្ហូបអាហារ ទឹកភ្លើង សាលារៀនកូនៗ និងថែទាំសុខភាព។
៣០% លើបំណងប្រាថ្នា (Wants): ការកម្សាន្ត ញ៉ាំអាហារក្រៅផ្ទះ ដំណើរកម្សាន្ត និងទិញសម្ភារៈផ្ទាល់ខ្លួន។
២០% លើការសន្សំ & វិនិយោគ (Savings): កញ្ចប់គ្រាអាសន្ន ការសងបំណុលមុនកាលកំណត់ និងការវិនិយោគរយៈពេលវែង។

គន្លឹះគ្រប់គ្រងហិរញ្ញវត្ថុគ្រួសារឆ្លាតវៃ

  • កត់ត្រារាល់ចំណាយតូចៗតាមទូរស័ព្ទភ្លាមៗបន្ទាប់ពីទិញទំនិញរួច។
  • ពិនិត្យមើលកញ្ចប់ថវិកាជារៀងរាល់ចុងសប្ដាហ៍ដើម្បីជៀសវាងការចំណាយលើស។
  • បង្វែរប្រាក់សន្សំចូលទៅក្នុងគណនីដាច់ដោយឡែក (High-interest savings) ភ្លាមៗពេលបើកប្រាក់ខែ។
  • ប្រើប្រាស់រូបិយប័ណ្ណរៀល (KHR) សម្រាប់ការចំណាយប្រចាំថ្ងៃ និងដុល្លារ (USD) សម្រាប់សន្សំ ឬទិញទ្រព្យសកម្មធំៗ។
